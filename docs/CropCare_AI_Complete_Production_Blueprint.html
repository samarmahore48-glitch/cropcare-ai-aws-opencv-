1. Modernize Frontend Build & Offline PWA

Target Goal: Transform the frontend into an installable Progressive Web App (PWA) that loads in under 200 milliseconds, functions 100% offline in rural farm plots, and requires zero app store approvals.

Why This Matters For Small Farmers

In rural areas, farmers frequently experience 2G/3G connections or complete signal blackouts in the middle of their field. A traditional website that downloads a heavy bundle every visit will fail. By using Vite + Service Workers, the entire application shell, diagnosis library, and offline crop doctor remain fully functional without internet.

Step 2: Production FastAPI Service (backend/main.py)

Create your Python backend using FastAPI and Ultralytics YOLO:

# backend/main.py
from fastapi import FastAPI, UploadFile, File, Form, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from PIL import Image
import io
import imagehash
from ultralytics import YOLO

app = FastAPI(title="CropCare AI Inference Engine", version="2.0")

# Enable CORS for frontend domain
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_methods=["*"],
    allow_headers=["*"],
)

# Load pre-trained / fine-tuned YOLO model
model = YOLO("yolov8n-crop-diseases.pt")

# In-memory duplicate hash store (replace with Redis/Postgres in Step 4)
SEEN_HASHES = set()

@app.post("/api/v1/scan")
async def scan_crop(file: UploadFile = File(...), crop: str = Form("Tomato")):
    if not file.content_type.startswith("image/"):
        raise HTTPException(status_code=400, detail="Uploaded file is not an image")

    contents = await file.read()
    image = Image.open(io.BytesIO(contents)).convert("RGB")

    # 1. Duplicate Image Detection via Perceptual Hash
    img_hash = str(imagehash.dhash(image))
    is_duplicate = img_hash in SEEN_HASHES
    SEEN_HASHES.add(img_hash)

    # 2. Run YOLO Object Detection for Disease Lesions
    results = model(image)
    detections = []
    total_area_pixels = image.width * image.height
    affected_pixels = 0

    for r in results:
        for box in r.boxes:
            x1, y1, x2, y2 = box.xyxy[0].tolist()
            conf = float(box.conf[0])
            cls_id = int(box.cls[0])
            name = model.names[cls_id]

            box_area = (x2 - x1) * (y2 - y1)
            affected_pixels += box_area

            detections.append({
                "disease": name,
                "confidence": round(conf * 100, 1),
                "bbox": [round(x1), round(y1), round(x2), round(y2)]
            })

    affected_percentage = min(100.0, round((affected_pixels / total_area_pixels) * 100, 1))

    return {
        "status": "success",
        "is_duplicate": is_duplicate,
        "image_hash": img_hash,
        "crop": crop,
        "detections_count": len(detections),
        "detections": detections,
        "severity_score": min(100, int(affected_percentage * 1.4)),
        "affected_area_percentage": affected_percentage
    }

Production Blueprint · Module 3

3. Connect Live Meteorological Weather API

Target Goal: Feed real-time microclimate metrics (relative humidity, rainfall, dew-point, wind) directly into the disease spread algorithm using the free Open-Meteo API.

Why Weather Drives Disease Spread

Fungal spores (such as Alternaria solani and Phytophthora infestans) cannot germinate on dry leaves. They require ≥ 80% relative humidity or standing water droplets for 12 to 24 consecutive hours. Connecting hyper-local weather turns a static diagnosis into an early warning radar.

Step 1: The Open-Meteo API Endpoint

The Open-Meteo API is 100% free for commercial & non-commercial use, requires no API key, and offers 1km spatial resolution:

GET https://api.open-meteo.com/v1/forecast?latitude=22.7196&longitude=75.8577&hourly=temperature_2m,relativehumidity_2m,precipitation_probability,windspeed_10m&daily=temperature_2m_max,precipitation_probability_max&timezone=auto

Step 2: Frontend Weather Service (src/services/weatherService.ts)

// src/services/weatherService.ts
export interface LiveWeatherReport {
  temperature: number;
  humidity: number;
  rainProbability: number;
  windSpeed: number;
  fungalRiskLevel: 'Low' | 'Moderate' | 'Critical';
  advisory: string;
}

export async function fetchFieldWeather(lat: number, lng: number): Promise<LiveWeatherReport> {
  const url = `https://api.open-meteo.com/v1/forecast?latitude=${lat}&longitude=${lng}&hourly=temperature_2m,relativehumidity_2m,precipitation_probability,windspeed_10m¤t_weather=true&timezone=auto`;

  const res = await fetch(url);
  if (!res.ok) throw new Error('Weather API fetch failed');
  const data = await res.json();

  const currentTemp = data.current_weather.temperature;
  const currentWind = data.current_weather.windspeed;
  const currentHumidity = data.hourly.relativehumidity_2m[0];
  const currentRainProb = data.hourly.precipitation_probability[0] || 0;

  // Scientific Fungal Spore Germination Risk Logic
  let risk: 'Low' | 'Moderate' | 'Critical' = 'Low';
  let advisory = 'Environmental conditions are stable for crop growth.';

  if (currentHumidity >= 85 && currentRainProb >= 60) {
    risk = 'Critical';
    advisory = 'High humidity (>85%) and rain expected. Critical risk of fungal spore germination!';
  } else if (currentHumidity >= 75 || currentRainProb >= 40) {
    risk = 'Moderate';
    advisory = 'Elevated leaf moisture detected. Inspect lower canopy foliage.';
  }

  return {
    temperature: currentTemp,
    humidity: currentHumidity,
    rainProbability: currentRainProb,
    windSpeed: currentWind,
    fungalRiskLevel: risk,
    advisory
  };
}

Step 3: Calculating 7-Day Disease Spread Progression

Use this mathematical model to project how severity expands over 7 days based on current humidity:

// Calculates 7-day severity trajectory
export function calculateSpreadTrajectory(baseSeverity: number, humidity: number, rainProb: number) {
  // Humidity multiplier: >80% humidity accelerates growth by up to 1.8x
  const humidityFactor = humidity > 80 ? 1 + (humidity - 80) * 0.03 : 1.0;
  const rainFactor = 1 + (rainProb / 100) * 0.2;

  const trajectory = [];
  let current = baseSeverity;

  for (let day = 1; day <= 7; day++) {
    const dailyGrowth = current * 0.08 * humidityFactor * rainFactor;
    current = Math.min(95, Math.round(current + dailyGrowth));
    trajectory.push({ day: `Day ${day}`, severity: current });
  }

  return trajectory;
}

Step 3: Client-Side WebP Compression & Cloudflare R2 Upload

Never upload raw 5MB camera photos over rural 3G networks. Compress images to lightweight WebP (<200KB) in the browser before sending:

// Compresses canvas image to WebP blob
export async function compressLeafImage(file: File): Promise<Blob> {
  return new Promise((resolve) => {
    const img = new Image();
    img.src = URL.createObjectURL(file);
    img.onload = () => {
      const canvas = document.createElement('canvas');
      const MAX_WIDTH = 1200;
      const scale = MAX_WIDTH / img.width;
      canvas.width = MAX_WIDTH;
      canvas.height = img.height * scale;

      const ctx = canvas.getContext('2d')!;
      ctx.drawImage(img, 0, 0, canvas.width, canvas.height);

      canvas.toBlob((blob) => resolve(blob!), 'image/webp', 0.82);
    };
  });
}

6. Voice Input/Output (Bhashini / Web Speech)

Target Goal: Enable farmers to speak their crop questions and hear answers spoken aloud in their native tongue without typing.

Why Voice is Essential for Small Farmers

Many smallholder farmers have limited formal literacy or have wet, muddy hands while inspecting crops in the field. Voice interaction makes CropCare AI as natural as talking to an agricultural expert standing beside them.

Method A: Zero-Cost Native Web Speech API (Works on all Android Phones)

No backend, server, or API key needed. Runs directly on device:

// src/hooks/useVoiceAssistant.ts
import { useState, useEffect } from 'react';

export function useVoiceAssistant(languageCode = 'hi-IN') {
  const [isListening, setIsListening] = useState(false);
  const [transcript, setTranscript] = useState('');

  // 1. Speech-to-Text (Farmer Speaking)
  const startListening = () => {
    const SpeechRecognition = (window as any).SpeechRecognition || (window as any).webkitSpeechRecognition;
    if (!SpeechRecognition) {
      alert('Speech recognition is not supported in this browser.');
      return;
    }

    const recognition = new SpeechRecognition();
    recognition.lang = languageCode; // e.g. 'hi-IN', 'mr-IN', 'te-IN', 'ta-IN'
    recognition.continuous = false;
    recognition.interimResults = false;

    recognition.onstart = () => setIsListening(true);
    recognition.onend = () => setIsListening(false);
    recognition.onresult = (event: any) => {
      const text = event.results[0][0].transcript;
      setTranscript(text);
    };

    recognition.start();
  };

  // 2. Text-to-Speech (AI Doctor Speaking Back)
  const speakText = (text: string) => {
    if (!('speechSynthesis' in window)) return;
    window.speechSynthesis.cancel(); // Stop ongoing speech

    const utterance = new SpeechSynthesisUtterance(text);
    utterance.lang = languageCode;
    utterance.rate = 0.95; // Slightly slower for clear agricultural comprehension
    window.speechSynthesis.speak(utterance);
  };

  return { isListening, transcript, startListening, speakText };
}

Method B: Digital India's Bhashini AI (Government Open Speech Engine)

For rural dialects, Government of India's Bhashini AI offers the highest accuracy across Indian languages:

# backend/bhashini_service.py
import requests

BHASHINI_ENDPOINT = "https://dhruva-api.bhashini.gov.in/services/inference/pipeline"
API_KEY = "YOUR_BHASHINI_API_KEY"

def translate_and_speak(text: str, source_lang: str = "en", target_lang: str = "hi"):
    headers = {
        "Authorization": API_KEY,
        "Content-Type": "application/json"
    }
    payload = {
        "pipelineTasks": [
            {
                "taskType": "translation",
                "config": {
                    "language": { "sourceLanguage": source_lang, "targetLanguage": target_lang }
                }
            },
            {
                "taskType": "tts",
                "config": {
                    "language": { "sourceLanguage": target_lang },
                    "gender": "female"
                }
            }
        ],
        "inputData": { "input": [{ "source": text }] }
    }
    response = requests.post(BHASHINI_ENDPOINT, json=payload, headers=headers)
    return response.json()
